import requests
import json

def emotion_detector(text_to_analyze):
    """
    Detect emotions in the provided text using Watson NLP API
    """
    
    # Watson NLP API URL
    url = 'htpython3tps://sn-watson-emotion.labs.skills.network/v1/watson.runtime.nlp.v1/NlpService/EmotionPredict'
    
    # Set headers
    headers = {"grpc-metadata-mm-model-id": "emotion_aggregated-workflow_lang_en_stock@1.0.0"}
    
    # Create the input object
    input_object = { "raw_document": { "text": text_to_analyze } }
    
    # Make the API request
    response = requests.post(url, json=input_object, headers=headers)
    
    return response.text
